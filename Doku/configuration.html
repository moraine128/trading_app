System-Einstellungen und Parameter

config.json

{
  "trading": {
    "universes": ["DAX", "SP500"],
    "initial_capital": 10000,
    "max_positions": 10,
    "position_size": 1000
  },
  "risk_management": {
    "take_profit_pct": 0.05,
    "stop_loss_pct": 0.03,
    "trailing_stop_activation": 0.03,
    "trailing_stop_distance": 0.015
  },
  "indicators": {
    "rsi_period": 14,
    "macd_fast": 12,
    "macd_slow": 26,
    "macd_signal": 9
  }
}
