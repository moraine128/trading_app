Backtest via API

import requests

config = {
    "start_date": "2024-01-01",
    "end_date": "2025-12-31",
    "initial_capital": 10000,
    "universe": "DAX"
}

response = requests.post("http://localhost:8000/api/backtest", json=config)
results = response.json()
print(f"Return: {results['total_return_pct']:.2f}%")
